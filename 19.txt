from pathlib import Path

p = Path("shim/m11.c")
s = p.read_text()

old = """idr0 = rd32(SMMU_BASE + SMMU_IDR0);
        idr1 = rd32(SMMU_BASE + SMMU_IDR1);"""

new = """step("2a antes IDR0");
        idr0 = rd32(SMMU_BASE + SMMU_IDR0);
        showhex("smmu idr0", idr0);

        step("2b antes IDR1");
        idr1 = rd32(SMMU_BASE + SMMU_IDR1);
        showhex("smmu idr1", idr1);"""

if old not in s:
    raise SystemExit("ERRO: trecho IDR0/IDR1 não encontrado")

s = s.replace(old, new, 1)

replacements = {
    'showhex("smmu scr0", rd32(SMMU_BASE + SMMU_SCR0));':
        'step("2c antes SCR0"); showhex("smmu scr0", rd32(SMMU_BASE + SMMU_SCR0));',
    'showhex("smmu sgfsr", rd32(SMMU_BASE + SMMU_SGFSR));':
        'step("2d antes SGFSR"); showhex("smmu sgfsr", rd32(SMMU_BASE + SMMU_SGFSR));',
    'showhex("smr0", rd32(SMMU_BASE + SMMU_SMR(0)));':
        'step("2e antes SMR0"); showhex("smr0", rd32(SMMU_BASE + SMMU_SMR(0)));',
    'showhex("s2cr0", rd32(SMMU_BASE + SMMU_S2CR(0)));':
        'step("2f antes S2CR0"); showhex("s2cr0", rd32(SMMU_BASE + SMMU_S2CR(0)));',
    'showhex("cb0 sctlr", rd32(SMMU_BASE + smmu_pages * smmu_page + SMMU_CB_SCTLR));':
        'step("2g antes CB0 SCTLR"); showhex("cb0 sctlr", rd32(SMMU_BASE + smmu_pages * smmu_page + SMMU_CB_SCTLR));',
}

for old_line, new_line in replacements.items():
    if old_line not in s:
        print("AVISO: linha não encontrada:", old_line)
    else:
        s = s.replace(old_line, new_line, 1)

p.write_text(s)
print("instrumentação aplicada")
PY

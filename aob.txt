private async void lczxy7CustomCheckBox11_CheckedChanged(object sender, EventArgs e)
{
    string[] alvos = new string[] { "HD-Player" };
    if (!cosmicMem.SetProcess(alvos))
    {
        Console.Beep(300, 300); // Erro: Processo não encontrado
        return;
    }

    string[] search = {
        "DC 52 39 BD 27 C1 8B 3C C0 D0 F8 B9",
        "63 71 B0 BD 90 98 74 BB 00 00 80 B3",
        "5D C1 AB 2C 09 04 FF 18 EF E5 11 59"
    };

    string[] replace = {
        "00 00 00 3E 0A D7 23 3D D2 A5 F9 BC",
        "CD DC 79 44 90 98 74 BB 00 00 80 B3",
        "CD DC 79 44 58 34 09 BB B0 60 BE BA"
    };

    // Se desmarcou, restaura e sai fora direto
    if (!lczxy7CustomCheckBox11.Checked)
    {
        foreach (var patch in hsPeitoPatches)
        {
            cosmicMem.AobReplace(patch.Address, patch.OriginalBytes);
        }
        hsPeitoPatches.Clear();
        Console.Beep(500, 100); // Desativado com sucesso
        return;
    }

    // Se marcou, faz o scan e aplica
    hsPeitoPatches.Clear();
    bool sucesso = false;

    for (int i = 0; i < search.Length; i++)
    {
        var addrs = await cosmicMem.AoBScan(search[i]);
        if (addrs == null || !addrs.Any()) continue;

        foreach (var addr in addrs)
        {
            hsPeitoPatches.Add((addr, search[i]));
            cosmicMem.AobReplace(addr, replace[i]);
            sucesso = true;
        }
    }

    // Feedback sonoro dependendo do resultado
    if (sucesso)
    {
        Console.Beep(1000, 150); // Sucesso: Injetado corretamente
    }
    else
    {
        Console.Beep(300, 300);  // Erro: Nenhuma AOB encontrada
    }
}

from pathlib import Path

p = Path("shim/m11.c")
s = p.read_text()

marker = """static uint8_t gmu_dump_mem[GMU_MEM_SIZE] __attribute__((aligned(4096)));
"""

insert = """static uint8_t gmu_dump_mem[GMU_MEM_SIZE] __attribute__((aligned(4096)));

/*
 * Buffers reservados para o futuro teste PAS.
 * Nesta etapa eles não são enviados ao TrustZone.
 */
static uint8_t pas_metadata[0x2000] __attribute__((aligned(4096)));
static uint8_t pas_firmware[0x4000] __attribute__((aligned(4096)));
"""

if "static uint8_t pas_metadata" in s:
    raise SystemExit("Os buffers PAS já foram adicionados.")

if marker not in s:
    raise SystemExit("Marcador não encontrado; nenhum arquivo foi alterado.")

p.write_text(s.replace(marker, insert, 1))
print("buffers PAS adicionados; nenhum SMC foi chamado")
PY

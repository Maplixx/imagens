from pathlib import Path

p = Path("work/dt/live-twrp.dts")
s = p.read_text(errors="replace")
lines = s.splitlines()

targets = ("qcom,kgsl-iommu", "qcom,gmu", "gpucc", "gpu_cc")

for i, line in enumerate(lines):
    if not any(t in line.lower() for t in targets):
        continue

    start = i
    while start > 0 and lines[start].strip() not in ("{",):
        start -= 1

    # sobe até a linha que contém o nome do nó
    if start > 0:
        start -= 1

    depth = 0
    end = None
    for j in range(start, len(lines)):
        depth += lines[j].count("{")
        depth -= lines[j].count("}")
        if j > start and depth <= 0:
            end = j
            break

    print(f"\n===== bloco próximo da linha {i+1} =====")
    print("\n".join(f"{n+1}: {lines[n]}" for n in range(start, min(end or i+30, len(lines)))))
